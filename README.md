```
# --- 1. 환경 설정 및 데이터 저장 ---
# 최대 메모리를 30바이트로 제한 (키/값 길이를 합쳐 30 이상이면 LRU 정책 발동)
CONFIG SET maxmemory 30
# "user:1" (6글자) + "Alice" (5글자) = 11바이트 사용
SET user:1 "Alice"
# "user:2" (6글자) + "Bob" (3글자) = 9바이트 사용 (누적: 20)
SET user:2 "Bob"
# "user:3" (6글자) + "Charlie" (7글자) = 13바이트 사용 (누적: 33) 
# -> 30바이트 초과 발생! 가장 오래된 'user:1'이 LRU에 의해 자동으로 삭제됨

# --- 2. LRU 및 메모리 상태 확인 ---
# 'user:1'은 이미 삭제되었으므로 (nil) 반환. 'user:2'가 가장 최근 사용된 것으로 갱신됨
GET user:1
# 현재 메모리 사용량과 삭제된 키(evicted_keys) 개수 확인
INFO memory
# 현재 남아있는 키 목록 확인 ('user:1'이 삭제되었는지 검증 가능)
KEYS

# --- 3. TTL (만료 시간) 테스트 ---
# 새 데이터 저장
SET user:4 "David"
# 'user:4'에 5초의 유효기간 설정 (5초 뒤 자동 삭제 예약)
EXPIRE user:4 5
# 5초 이내에 호출하면 남은 시간(초)을 반환
TTL user:4

# --- 4. 만료 후 조회 ---
# 5초 대기 후 조회하면 이미 만료되었으므로 삭제됨 (nil 반환)
GET user:4
# 이미 삭제되었으므로 존재하지 않는 키를 의미하는 -2 반환
TTL user:4
```
